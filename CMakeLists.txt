cmake_minimum_required(VERSION 3.16)
project(kindle_dashboard VERSION 0.1.0 LANGUAGES CXX)
option(DASHBOARD_STATIC "Link all runtime libraries statically for Kindle" OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
find_package(OpenSSL REQUIRED)
find_path(FREETYPE_INCLUDE ft2build.h PATH_SUFFIXES freetype2 HINTS "${CMAKE_SOURCE_DIR}/.build/include" REQUIRED)
find_path(PNG_INCLUDE png.h HINTS "${CMAKE_SOURCE_DIR}/.build/include" REQUIRED)
find_path(CURL_INCLUDE curl/curl.h HINTS "${CMAKE_SOURCE_DIR}/.build/include" REQUIRED)
find_path(YAML_INCLUDE yaml.h HINTS "${CMAKE_SOURCE_DIR}/.build/include" REQUIRED)
find_path(JSON_INCLUDE nlohmann/json.hpp HINTS "${CMAKE_SOURCE_DIR}/.build/include" REQUIRED)
find_library(FT_LIB NAMES freetype libfreetype.so.6 REQUIRED)
find_library(PNG_LIB NAMES png16 png libpng16.so.16 REQUIRED)
find_library(CURL_LIB NAMES curl libcurl.so.4 REQUIRED)
find_library(YAML_LIB NAMES yaml libyaml-0.so.2 REQUIRED)
find_path(HB_INCLUDE hb.h PATH_SUFFIXES harfbuzz HINTS "${CMAKE_SOURCE_DIR}/.build/include" REQUIRED)
find_library(HB_LIB NAMES harfbuzz libharfbuzz.so.0 REQUIRED)
add_library(dashboard_core src/core.cpp src/render.cpp src/data.cpp)
target_include_directories(dashboard_core PUBLIC src ${FREETYPE_INCLUDE} ${PNG_INCLUDE} ${CURL_INCLUDE} ${YAML_INCLUDE} ${JSON_INCLUDE} ${HB_INCLUDE})
target_link_libraries(dashboard_core PUBLIC ${FT_LIB} ${PNG_LIB} ${CURL_LIB} ${YAML_LIB} ${HB_LIB} OpenSSL::Crypto Threads::Threads)
if(DASHBOARD_STATIC)
    target_compile_definitions(dashboard_core PUBLIC DASHBOARD_MUSL_THREADING=1)
    set(ZLIB_USE_STATIC_LIBS ON)
    find_package(ZLIB REQUIRED)
    target_link_libraries(dashboard_core PUBLIC OpenSSL::SSL OpenSSL::Crypto ZLIB::ZLIB)
    target_link_options(dashboard_core PUBLIC -static -Wl,-z,stack-size=1048576)
endif()
target_compile_options(dashboard_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(kindle-dashboard src/main.cpp)
target_link_libraries(kindle-dashboard dashboard_core)
option(DASHBOARD_WINDOW "Build the native X11/WSLg preview window (development only)" OFF)
if(DASHBOARD_WINDOW)
    find_path(X11_INCLUDE X11/Xlib.h HINTS "${CMAKE_SOURCE_DIR}/.build/x11/usr/include" REQUIRED)
    find_library(X11_LIB NAMES X11 libX11.so.6 REQUIRED)
    target_include_directories(kindle-dashboard PRIVATE ${X11_INCLUDE})
    target_link_libraries(kindle-dashboard ${X11_LIB})
    target_compile_definitions(kindle-dashboard PRIVATE DASHBOARD_WINDOW=1)
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(dashboard-check tests/check.cpp)
    target_link_libraries(dashboard-check dashboard_core)
    add_test(NAME dashboard-check COMMAND dashboard-check ${CMAKE_SOURCE_DIR})
    if(DASHBOARD_WINDOW)
        add_executable(dashboard-window-check tests/window.cpp)
        target_include_directories(dashboard-window-check PRIVATE ${X11_INCLUDE})
        target_link_libraries(dashboard-window-check dashboard_core ${X11_LIB})
        if(DEFINED ENV{DISPLAY})
            add_test(NAME dashboard-window-check COMMAND dashboard-window-check)
        endif()
    endif()
endif()
install(TARGETS kindle-dashboard DESTINATION bin)
install(DIRECTORY assets DESTINATION .)
